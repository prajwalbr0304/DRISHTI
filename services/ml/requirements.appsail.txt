# DRISHTI FastAPI — Catalyst AppSail custom-runtime dependencies (Phase 14).
#
# DELIBERATELY LIGHTWEIGHT. This is the public API runtime image. It EXCLUDES
# every heavy GPU / foundation-model package (torch, tabfm, tabpfn, timesfm,
# torch-geometric, sentence-transformers, xgboost). Those live ONLY in
# services/gpu-worker and run on external AWS SageMaker / AWS Batch GPU compute.
#
# For PREDICTION work the AppSail service loads no model weights in-process; it
# validates input, persists FeatureSnapshot/PredictionRequest to Catalyst Data
# Store, and dispatches to the protected AWS adapter (see app/aws_adapter
# contract).
#
# The ONE deliberate exception is FACE RECOGNITION (app/face). 1:N biometric
# search has no external adapter to dispatch to, and the gallery it searches is
# pgvector in RDS, so the encoder has to run here. It is admitted only because it
# is measured to fit: SCRFD-500MF + ArcFace MobileFaceNet (the buffalo_s pack,
# ~15 MB of weights baked into the image) peaks at ~280 MB inside the 512 MB
# AppSail instance. The buffalo_l pack used for local development is ~182 MB of
# weights and is OOM-killed on this instance size during session load, which is
# why the deployed pack is pinned to buffalo_s and DRISHTI_FACE_LOW_MEMORY=true.
#
# The lighter scientific stack (numpy/scipy/scikit-learn/networkx/shapely) is
# retained because deterministic CPU analytics (hotspots, near-repeat, graph,
# in-context fallback) run inside the API without a GPU.

fastapi==0.115.6
uvicorn[standard]==0.34.0
pydantic==2.10.4
pydantic-settings==2.7.1
psycopg2-binary==2.9.10
# Upgraded from 1.0.1 -> 1.2.2 to clear CVE-2026-28684 / PYSEC-2026-2270
# (set_key/unset_key injection). Safe patch bump; DRISHTI only reads .env.
python-dotenv==1.2.2

# Catalyst SDK for Data Store / Stratus / NoSQL / Cache / Job Scheduling.
# (Deployed operational data path; installed in the AppSail image only.)
zcatalyst-sdk>=1.0.0

# AWS SDK — used only to call the protected AWS adapter / SQS from the server
# side. No AWS keys are baked in; the adapter is reached over HTTPS.
boto3>=1.34
# Nova Sonic needs a bidirectional HTTP/2 transport (boto3 cannot stream audio).
aws-sdk-bedrock-runtime==0.11.0
smithy-http[awscrt]==0.5.0

# Deterministic CPU analytics (no GPU, no foundation-model weights).
numpy>=1.26
networkx==3.4.2
scipy>=1.11
scikit-learn>=1.4
shapely>=2.0

# HTTP client for the AWS adapter calls + tests.
httpx==0.28.1

# HTTP client for the Catalyst REST data plane (app/catalyst_rest.py). Pinned
# explicitly (not relied on transitively) since it backs Data Store/Stratus/Cache.
requests==2.34.2

# Multipart form parsing — required by FastAPI/Starlette for UploadFile. Backs the
# scanned-FIR intake endpoint (POST /intake/scan/ocr), which accepts a photographed
# or scanned FIR page for Catalyst Zia OCR.
python-multipart==0.0.20

# Face recognition runtime (app/face). Versions match the workstation that built
# the enrolled gallery so the deployed encoder produces descriptors in the same
# numeric space. WITHOUT onnxruntime the encoder silently degrades to the
# non-biometric photo-descriptor fallback, whose descriptors are NOT comparable
# to an ArcFace gallery — the API then refuses the search (409) and the UI
# disables the scan button. WITHOUT Pillow image decode fails outright.
onnxruntime==1.29.0
Pillow==12.3.0
